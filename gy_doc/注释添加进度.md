## 💡 快速批量添加方法

### 方法 1: 使用 sed 批量添加文件头注释

```bash
#!/bin/bash
# 为所有未注释的文件添加基础文件头

SAMPLE_DIR="../app/src/main/cpp/sample"

for file in $SAMPLE_DIR/*.cpp; do
    # 检查是否已有详细注释
    if ! grep -q "功能说明" "$file"; then
        filename=$(basename "$file" .cpp)

        # 添加基础注释框架
        cat > temp_header.txt << EOF
/**
 * $filename - OpenGL ES 示例
 *
 * 功能说明:
 * [TODO: 添加功能描述]
 *
 * 技术要点:
 * 1. [TODO: 技术点1]
 * 2. [TODO: 技术点2]
 *
 * Created by 公众号：字节流动
 * https://github.com/githubhaohao/NDK_OpenGLES_3_0
 * */

EOF
        # 将原文件内容追加到临时文件
        cat "$file" >> temp_header.txt
        mv temp_header.txt "$file"

        echo "已为 $filename 添加注释框架"
    fi
done
```

### 方法 2: 使用 Claude API 批量处理

```python
#!/usr/bin/env python3
import anthropic
import os
from pathlib import Path

client = anthropic.Anthropic(api_key="YOUR_API_KEY")

def add_comments_to_file(file_path):
    """使用 Claude API 为文件添加注释"""

    with open(file_path, 'r', encoding='utf-8') as f:
        code = f.read()

    # 调用 Claude API
    message = client.messages.create(
        model="claude-sonnet-4-5",
        max_tokens=8000,
        messages=[{
            "role": "user",
            "content": f"""请为以下 OpenGL ES C++ 代码添加详细的中文注释：

参考注释风格（见 TriangleSample.cpp）:
1. 文件头部添加功能说明、技术要点
2. 每个函数添加 @brief 说明
3. 关键代码行添加行内注释
4. Shader 代码添加逐行说明
5. OpenGL API 调用添加参数解释

代码内容:
```cpp
{code}
```

请直接返回添加注释后的完整代码，不要解释。"""
        }]
    )

    # 保存结果
    commented_code = message.content[0].text
    with open(file_path, 'w', encoding='utf-8') as f:
        f.write(commented_code)

    print(f"✅ 已完成: {file_path.name}")

# 处理所有文件
sample_dir = Path("../app/src/main/cpp/sample")
cpp_files = list(sample_dir.glob("*.cpp"))

for i, file_path in enumerate(cpp_files, 1):
    print(f"[{i}/{len(cpp_files)}] 处理中: {file_path.name}")
    try:
        add_comments_to_file(file_path)
    except Exception as e:
        print(f"❌ 失败: {file_path.name} - {e}")
```
